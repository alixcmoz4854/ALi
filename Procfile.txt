def get_db():
    conn = sqlite3.connect("panel_ultimate.db", check_same_thread=False)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("CREATE TABLE IF NOT EXISTS users (user_id INTEGER PRIMARY KEY, balance INTEGER DEFAULT 0)")
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS panels (
                panel_id INTEGER PRIMARY KEY AUTOINCREMENT, 
                user_id INTEGER, 
                panel_type TEXT, 
                username TEXT, 
                password TEXT, 
                total_gb INTEGER, 
                used_gb INTEGER DEFAULT 0, 
                user_count INTEGER DEFAULT 0, 
                status TEXT DEFAULT 'فعال'
            )
        """)
        conn.commit()

def get_user(user_id):
    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT balance FROM users WHERE user_id = ?", (user_id,))
        row = cursor.fetchone()
        if not row:
            cursor.execute("INSERT INTO users (user_id, balance) VALUES (?, 0)", (user_id,))
            conn.commit()
            return 0
        return row['balance']

def update_balance(user_id, amount):
    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("UPDATE users SET balance = balance + ? WHERE user_id = ?", (amount, user_id))
        conn.commit()

def add_panel(user_id, panel_type, username, password, total_gb):
    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("INSERT INTO panels (user_id, panel_type, username, password, total_gb) VALUES (?, ?, ?, ?, ?)", 
                       (user_id, panel_type, username, password, total_gb))
        conn.commit()

def get_user_panels(user_id):
    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT panel_id, panel_type, username, password, total_gb, used_gb, user_count FROM panels WHERE user_id = ?", (user_id,))
        return cursor.fetchall()

def get_all_users():
    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("SELECT user_id FROM users")
        return [r['user_id'] for r in cursor.fetchall()]

async def send_broadcast(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if update.effective_user.id != ADMIN_ID:
        return ConversationHandler.END
    msg_text = update.message.text
    users = get_all_users()
    success, failed = 0, 0
    await update.message.reply_text("⏳ در حال ارسال پیام...")
    for uid in users:
        try:
            await context.bot.send_message(chat_id=uid, text=f"📢 **اطلاعیه:**\n\n{msg_text}", parse_mode="Markdown")
            success += 1
            await asyncio.sleep(0.04)
        except Exception:
            failed += 1
    await update.message.reply_text(f"🚀 **ارسال همگانی انجام شد.**\n\n✅ موفق: `{success}`\n❌ ناموفق: `{failed}`", parse_mode="Markdown")
    return ConversationHandler.END
